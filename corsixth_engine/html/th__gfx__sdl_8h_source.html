205
208 inline void push_pixel(uint32_t iValue) {
209 if (y < height) {
210 store_argb(iValue);
211 x++;
212 if (x >= width) {
213 x = 0;
214 y++;
215 }
216 } else {
217 throw std::logic_error("Attempt to push_pixel past the end of the image");
